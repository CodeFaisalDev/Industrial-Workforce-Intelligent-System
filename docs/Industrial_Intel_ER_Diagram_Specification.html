1. EXECUTIVE SUMMARY: RELATIONAL DATABASE ARCHITECTURE & DESIGN PRINCIPLES
Industrial Intel v2.0 | Architectural Specification

Multi-Tenant Core & Storage Strategy

The Industrial Intel database layer is engineered on PostgreSQL 16 utilizing Neon Serverless Postgres with the pgvector extension. It provides absolute corporate tenant isolation (anchored at companies.id) while allowing high-speed client-side edge synchronization through browser IndexedDB cache terminals.

Zero-Loss Edge Architecture: Kiosks deployed on factory floor entry gates record punches locally in IndexedDB when internet connectivity drops. The client-side cache stores unsynced = 0 records and auto-syncs over REST to biometric_logs upon reconnection.

Vector Indexing & Facial Recognition Math

Facial templates are encoded into 128-dimensional floating point vectors stored in face_embeddings.embedding. The database utilizes Euclidean L2 distance for biometric matches:

SELECT employee_id FROM face_embeddings WHERE is_active = TRUE ORDER BY embedding <-> $1 LIMIT 1;
-- Validated Threshold: Euclidean L2 Distance <= 0.05

9. POSTGRESQL DDL SCRIPT & VECTOR EXTENSION DEFINITION
Production Migration Script | lib/init-db.ts & migrate.js
-- 1. Enable Required Database Extensions
CREATE EXTENSION IF NOT EXISTS vector;
CREATE EXTENSION IF NOT EXISTS pg_trgm;

-- 2. Multi-Tenancy & Core Organizational Tables
CREATE TABLE IF NOT EXISTS companies (
  id SERIAL PRIMARY KEY,
  name VARCHAR(255) UNIQUE NOT NULL,
  created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS departments (
  id SERIAL PRIMARY KEY,
  name VARCHAR(255) NOT NULL,
  manager_id INTEGER,
  company_id INTEGER REFERENCES companies(id) ON DELETE CASCADE,
  created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  UNIQUE(name, company_id)
);

CREATE TABLE IF NOT EXISTS employees (
  id SERIAL PRIMARY KEY,
  name VARCHAR(255) NOT NULL,
  email VARCHAR(255) UNIQUE NOT NULL,
  password VARCHAR(255),
  role VARCHAR(50) NOT NULL DEFAULT 'Worker',
  department_id INTEGER REFERENCES departments(id) ON DELETE SET NULL,
  company_id INTEGER REFERENCES companies(id) ON DELETE CASCADE,
  status VARCHAR(50) NOT NULL DEFAULT 'Active',
  created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS manager_worker_access (
  id SERIAL PRIMARY KEY,
  manager_id INTEGER NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  worker_id INTEGER NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  UNIQUE(manager_id, worker_id)
);

-- 3. Biometrics & Attendance Logs
CREATE TABLE IF NOT EXISTS face_embeddings (
  id SERIAL PRIMARY KEY,
  employee_id INTEGER NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  embedding vector(128) NOT NULL,
  enrolled_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP,
  is_active BOOLEAN DEFAULT TRUE
);
CREATE TABLE IF NOT EXISTS biometric_logs (
  id SERIAL PRIMARY KEY,
  employee_id INTEGER NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  timestamp TIMESTAMPTZ NOT NULL,
  log_type VARCHAR(50) NOT NULL,
  device_id VARCHAR(255) NOT NULL,
  verified_by_face BOOLEAN DEFAULT FALSE,
  gps_lat DOUBLE PRECISION,
  gps_lng DOUBLE PRECISION,
  created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 4. Operations, Scheduling & Leave
CREATE TABLE IF NOT EXISTS shifts (
  id SERIAL PRIMARY KEY,
  employee_id INTEGER NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  date DATE NOT NULL,
  start_time TIME NOT NULL,
  end_time TIME NOT NULL,
  status VARCHAR(50) NOT NULL DEFAULT 'Scheduled',
  created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

CREATE TABLE IF NOT EXISTS leave_requests (
  id SERIAL PRIMARY KEY,
  employee_id INTEGER NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  leave_type VARCHAR(50) NOT NULL,
  start_date DATE NOT NULL,
  end_date DATE NOT NULL,
  reason TEXT NOT NULL,
  status VARCHAR(50) NOT NULL DEFAULT 'Pending',
  created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);

-- 5. Deterministic Payroll Ledgers
CREATE TABLE IF NOT EXISTS payroll_ledgers (
  id SERIAL PRIMARY KEY,
  employee_id INTEGER NOT NULL REFERENCES employees(id) ON DELETE CASCADE,
  period_start DATE NOT NULL,
  period_end DATE NOT NULL,
  regular_hours DOUBLE PRECISION DEFAULT 0,
  overtime_hours DOUBLE PRECISION DEFAULT 0,
  gross_pay DOUBLE PRECISION DEFAULT 0,
  deductions DOUBLE PRECISION DEFAULT 0,
  net_pay DOUBLE PRECISION DEFAULT 0,
  status VARCHAR(50) DEFAULT 'Draft',
  created_at TIMESTAMPTZ DEFAULT CURRENT_TIMESTAMP
);
